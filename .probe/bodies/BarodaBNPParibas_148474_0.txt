                                       <h3>Explanation of IR</h3>
                                       <p>Information Ratio (IR) is a risk adjusted return metric which helps investors understand whether the Scheme has generated excess return over the benchmark index consistently. It is often used as a measure of a portfolio manager's level of skill and ability to generate excess returns, relative to a benchmark and attempts to identify the consistency of the performance by incorporating standard deviation into the calculation.</p>
                                       <p>When a portfolio's IR is high, it indicates that it is performing well, meaning that it consistently produces risk adjusted returns. In contrast, a low IR may indicate a volatile portfolio, which means the portfolio generates less predictable returns. 
                                       </p>
                                       <h3>
                                           Formula for calculation of IR
                                       </h3>
                                       
                                       <div style="font-size: 15px;">
                                          <em>Information Ratio (IR)</em> =
                                          <span style="display: inline-block; text-align: center; vertical-align: middle; margin-left: 8px;">
                                            <span style="border-bottom: 1px solid #000; display: block;">Excess Returns</span>
                                            <span style="display: block;">Tracking Error</span>
                                          </span>
                                        </div>
                                        
                                        <div style="font-size: 15px;">
                                          Where <em>Excess Returns</em> = <em>Portfolio rate of returns</em> − <em>Benchmark rate of returns</em><br>
                                          and <em>Tracking Error</em> = <em>Standard Deviation of Excess Returns</em>
                                        </div>
                                        
                                        <p>As per the methodology of calculation mentioned in the circular by SEBI regarding disclosure of IR, daily portfolio return shall be calculated using arithmetic function, and the standard deviation will be calculated based on these daily return values. IR would be calculated for 1-year, 3-year, 5-year, and 10-year horizon using this methodology. </p>

                                        <p>Ideally one would prefer a portfolio which earns stable returns, over one which generates less consistent returns. Information ratio becomes useful here. We’ll assume that the benchmark index has given an average return of 14%.  Here’s how Information Ratio can help make an investment decision - </p>

document.addEventListener('DOMContentLoaded', function() {
    const sections = document.querySelectorAll('.scroll-section');
    const navLinks = document.querySelectorAll('.scroll-link');

    // Intersection Observer for active tab switching
    const observer = new IntersectionObserver(entries => {
        entries.forEach(entry => {
            const id = entry.target.getAttribute('id');
            const navLink = document.querySelector(`a[href="#${id}"]`);
            if (entry.isIntersecting) {
                navLinks.forEach(link => link.classList.remove('active'));
                navLink.classList.add('active');
            }
        });
    }, {
        threshold: 0.5 // 50% visibility required to make section active
    });

    sections.forEach(section => observer.observe(section));

    // Smooth scrolling for anchor links
    navLinks.forEach(link => {
        link.addEventListener('click', function(e) {
            e.preventDefault();
            const targetId = this.getAttribute('href').substring(1);
            const targetSection = document.getElementById(targetId);
            targetSection.scrollIntoView({ behavior: 'smooth' });
        });
    });
});

function toggleFields() {
    const mode = document.getElementById("mode_swpreturn_5").value;
    
    
    // Toggle visibility
    const lumpsumFields = document.getElementById("lumpsumFields_5");
    const sipFields = document.getElementById("sipFields_5");

    if (mode === "Lumpsum") {
        lumpsumFields.style.display = "block";
        sipFields.style.display = "none";

        // Enable Lumpsum fields and disable SIP fields
        enableInputs(lumpsumFields);
        disableInputs(sipFields);
    } else if (mode === "SIP") {
        lumpsumFields.style.display = "none";
        sipFields.style.display = "block";

        // Enable SIP fields and disable Lumpsum fields
        enableInputs(sipFields);
        disableInputs(lumpsumFields);
    }
}

  function rollingreturnscta() {

    const period = $('#rolling_period').val();
    const startDate = $('#rolling_start_date').val();

        console.log('Please enter a Scheme AMFI Code.');
        $('#loader-div').hide();
        return false;
    }

    if (!period) {
        console.log('Please select a rolling return period.');
        $('#loader-div').hide();
        return false;
    }

    if (!startDate) {
        console.log('Please provide a valid start date.');
        $('#loader-div').hide();
        return false;
    }

    const data = {
        key: apiKey,
        period: period,
        scheme_amfi_code: schemeCode,
        start_date: startDate
    };

    const queryString = Object.keys(data)
        .map(key => `${key}=${encodeURIComponent(data[key])}`)
        .join('&');

    const url =
        `https://mfapi.advisorkhoj.com/baroda/getRollingReturnsVsBenchmark?${queryString}`;

    // Show loader only after validation passes
    $('#loader-div').show();

    $.ajax({
        url: url,
        type: 'POST',
        contentType: 'application/json',
        data: JSON.stringify(data),
        dataType: 'json',

        success: function (result) {
            if (
                result &&
                result.msg === 'Success' &&
                result.rolling_return_summary_list &&
                result.rolling_return_summary_list.length > 0
            ) {
                processRollingReturnsData(
                    result.rolling_return_summary_list
                );

                $('#NoData').html('');
            } else {
                if (myRollingChart) {
                    myRollingChart.destroy();
                    myRollingChart = null;
                }

                $('#NoData').html('No Data found');
            }
        },

        error: function () {
            $('#rolling-returns-results').html(
                '<p>An error occurred while fetching data. Please try again later.</p>'
            );

            if (myRollingChart) {
                myRollingChart.destroy();
                myRollingChart = null;
            }

            $('#NoData').html('No Data found');
        },

        // Always runs after success or error
        complete: function () {
            $('#loader-div').hide();
        }
    });

    return false;

    rangeInput.value = value;
    applyFill(rangeInput);

function applyFill(slider) {
  const percentage = 100 * (slider.value - slider.min) / (slider.max - slider.min);
  const bg = `linear-gradient(90deg, ${settings.fill} ${percentage}%, ${settings.background} ${percentage + 0.1}%)`;
  slider.style.background = bg;
}

<script type="text/javascript" nonce="12D3+PpKOkZI/vnMJk6dpg==">
$(document).on('ready',function(){
    $(document).on('click','.acc__title',function(e) {
        var dropDown = $(this).closest('.acc__card').find('.acc__panel');
        $(this).closest('.acc').find('.acc__panel').not(dropDown).slideUp();
        if ($(this).hasClass('active')) {
            $(this).removeClass('active');
        } else {
            $(this).closest('.acc').find('.acc__title.active').removeClass('active');
            $(this).addClass('active');
        }
        dropDown.stop(false, true).slideToggle();
        e.preventDefault();
    });

    var hash = window.location.hash;
    $(function() {
        $('.tabs-nav a').click(function(){
            window.location.hash = $(this).attr('aria-controls');
        });
        $('[href="'+hash+'"]').trigger('click');
    });
});
